#Collectivization - start
country_event = {
    id = 1512058
    title = "Collectivization in $COUNTRY$"
    desc = EVTDESC1512058

    trigger = {

        has_country_flag = collectivise_farms

        any_owned_province = {
                OR = {
                    trade_goods = fruit
                    trade_goods = grain
                    trade_goods = tea
                    trade_goods = tobacco
                    trade_goods = coffee
                    trade_goods = cotton
                }
                NOT = { has_province_modifier = collectivisation }

        }
    }

    option = {
        name = "This will increase crop yield!"

        any_owned = {
            limit = {
                OR = {
                    trade_goods = fruit
                    trade_goods = grain
                    trade_goods = tea
                    trade_goods = tobacco
                    trade_goods = coffee
                    trade_goods = cotton
                }
                NOT = { has_province_modifier = collectivisation }
            }
            add_province_modifier = { name = collectivisation duration = -1 }
        }

    }

}

province_event = {
    id = 1512059
    title = "Grain Hoarding and Speculation"
    desc = EVTDESC1512059

    trigger = {
        owner = {
            ruling_party_ideology = communist
            NOT = { has_country_modifier = liquidating_ruling_class }
        }
        has_pop_type = farmers
        NOT = { has_province_modifier = grain_hoarding_speculation }
    }

    mean_time_to_happen = {
        days = 5
    }

    option = {
        name = "We will seize this hoarded grain soon enough!"

        any_pop = {
            limit = { NOT = { strata = rich } }
            militancy = 2
        }
        
        add_province_modifier = { name = grain_hoarding_speculation duration = -1 }
    }
}
