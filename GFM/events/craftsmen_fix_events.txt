country_event = {
	id = 9999901
	title = "EVTNAME9999901"
	desc = "EVTDESC9999901"
	picture = "Opiumwar"

	is_triggered_only = yes
	fire_only_once = yes

	option = {
		name = "USE THIS OPTION FIRST"

		# For non-RUS: convert non-primary/non-accepted OR (African diaspora that is not primary)
		any_owned = {
			limit = {
				OR = {
					is_core = THIS
					is_colonial = no
				}
				NOT = { owned_by = RUS }
			}
			any_pop = {
				limit = {
					OR = {
						type = craftsmen
						type = clerks
					}
					OR = {
						AND = {
							is_primary_culture = no
							is_accepted_culture = no
						}
						AND = {
							is_culture_group = african_diaspora_cultures
							is_primary_culture = no
						}
					}
				}
				pop_type = farmers
			}
		}

		# For RUS: convert all non-primary (this already spares primary cultures, including diaspora ones)
		any_owned = {
			limit = {
				OR = {
					is_core = THIS
					is_colonial = no
				}
				owned_by = RUS
			}
			any_pop = {
				limit = {
					OR = {
						type = craftsmen
						type = clerks
					}
					is_primary_culture = no
				}
				pop_type = farmers
			}
		}
	}
}
