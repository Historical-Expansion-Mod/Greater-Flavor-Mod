# ============================================================
# music.txt - sound played when an event appears
#
# Format:   picture_name = "file.wav"
#
# - picture_name is the file name of the event picture, without
#   the extension (the one in gfx\pictures\events\).
# - file.wav is the sound, which must be in the sound\ folder
#   of your mod.
# - Only .wav files are supported (standard PCM). A renamed mp3
#   will not work.
# - Lines starting with # are comments.
#
# Example:
#   Utopia = "civil_war_usa.wav"
# ============================================================
